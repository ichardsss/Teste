git status
# Inicializar um repositorio GIT
git init .

# Mostrar o estado atual do repositorio
git status

# Adicionar um arquivo em stage
git add <nome_do_arquivo>

# Commitar e substituir o ultimo commit 
git commit -m "sua msg" --amend

# Adicionar endereço remoto em um repositorio local
git remote add <url>

# Enviar alterações e definir a upstream(remoto) com a branch local
git push -u origin <nome_da_branch>

# Enviar alterações para o github
git push
git push origin <nome_da_branch>

# Enviar alterações de uma branch local
# para uma branch remota

# Remover arquivo do stage 
git rm --cached<nome_do_arquivo>

# restaurar um arquivo
git restore <nome_do_arquivo>

# salvar o estado do repositorio
git commit -m "sua msg"

# mostrar o historico de commits
git log #alemona #git alemona 6662
git log --oneline

# mostrar as alterações no repositorio
git diff #git alemona

# exibir alterações de um commit
git show <hash>v

# listar as branchs
git branch

# renomear a branch
git branch -m <novo_nome>

# criar uma nova branch
git branch <nome_novo_branch>

# deletar uma branch
git branch -d <nome_branch>

# trocar branch 
git switch <nome_da_branch>

# mesclar uma branch em outra 
git merge <nome_da_branch

# mostrar autores de um arquivo 
 git blame <nome_do_arquivo>

# Voltar para um commit anterior (LOCAL)
git reset <hash_do_commit>

# Desfazer um commit (REMOTO)
git revert <hash_do_commit>

# Clonar repositorio
git clone <url_copiada>

# Verificar endereço remoto
git remote -v
